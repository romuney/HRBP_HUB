## 04a HH · отпуска — заменить код целиком (поставка 67: остаток по неосновным позициям — xdays; в справочной таблице — sec_yellow / sec_red)

```sql
-- ============================================================================
-- HH · отпуска (11.11, метрика «Накопленный отпуск»). Остаток накопленных дней отпуска сотрудника на конец
-- каждого слота — по ОСНОВНОЙ позиции (legal_position_rk строки витрины структуры), сумма по всем типам отпуска.
-- Только штатные (employment_relation_type_desc = 'Штатный сотрудник' — разрез staff = «Штат»): метрика считается
-- по штатным и в числителе, и в знаменателе, какой бы разрез ни выбрали (владелец 11.11).
--   Источник — prod_v_sse_crossdata.legal_position_vacation: бизнес-версии по (позиция, тип отпуска).
--   Версия на дату — состояние на КОНЕЦ дня: valid_from <= d 23:59:59 <= valid_to. valid_to включительный — следующая
--   версия начинается через 1 секунду (этап 0, запрос 2); срез «valid_from::date <= d < valid_to::date» терял бы
--   позицию в каждый день окончания версии (начисление 1-го числа — это каждый конец месяца). deleted_flg = 0.
--   Остаток — в сотых долях дня (целым: 7,33 → 733) — сумма по команде без потерь, чарт делит на 100. Зоны — по
--   точной сумме: жёлтая 28 ≤ дни < 56, красная ≥ 56.
-- Начало метрики (vac_start_dt) — первый месяц отчёта, где запись есть хотя бы у hh_param.vac_min_cov (85 %) штатных
-- всей компании; раньше — история отпусков не загружена, слоты пустые (не нули). Месяц после начала с покрытием
-- ниже порога не прячется — его печатает «HH · проверки».
--   hh_vac        сотрудник × дата слота (только штатные в численности, слоты отчёта idx >= 0): days — точная сумма,
--                 NULL — записи об отпуске нет (сотрудник в знаменателе и «не в зоне»); xdays — сумма по НЕОСНОВНЫМ
--                 позициям (12.11): позиции сотрудника на ту же дату из prod_v_emart.legal_position_d, кроме основной
--                 (legal_position_rk строки структуры; признак main_legal_position_flg с ней совпадает — этап 0б,
--                 запрос 3), NULL — неосновных позиций с записью об отпуске нет. Долю зон не меняет: штатные с xdays в
--                 зоне — отдельные компоненты vxy / vxr (пометка у метрики, владелец 11.11).
--   hh_vac_cov    покрытие по месяцам отчёта: штатных, из них с записью, доля
--   hh_vac_start  vac_start_dt — первый день месяца начала (5999-01-01 — порога не достиг ни один месяц)
-- ============================================================================
drop table if exists hh_vac;
create table hh_vac as
select s.mdm_employee_rk, s.business_dt, v.days, w.xdays
from hh_src s
left join (
    select q.mdm_employee_rk, q.business_dt, sum(x.accum_vacation_day)::numeric as days
    from (select distinct mdm_employee_rk, business_dt, legal_position_rk
          from hh_src
          where is_slot_row = 1 and hc = 1 and staff = 'Штат' and legal_position_rk is not null
            and business_dt in (select slot_end from hh_calendar where idx >= 0)) q
    join prod_v_sse_crossdata.legal_position_vacation x
        on x.legal_position_rk::text = q.legal_position_rk
       and x.valid_from_dttm <= q.business_dt + interval '1 day' - interval '1 second'
       and x.valid_to_dttm   >= q.business_dt + interval '1 day' - interval '1 second'
    where x.deleted_flg = 0
    group by 1, 2
) v
    on v.mdm_employee_rk = s.mdm_employee_rk and v.business_dt = s.business_dt
left join (
    select q.mdm_employee_rk, q.business_dt, sum(x.accum_vacation_day)::numeric as xdays
    from (select distinct s2.mdm_employee_rk, s2.business_dt, d.legal_position_rk::text as rk
          from hh_src s2
          join prod_v_emart.legal_position_d d
              on d.mdm_employee_rk = s2.mdm_employee_rk and d.business_dt = s2.business_dt
          where s2.is_slot_row = 1 and s2.hc = 1 and s2.staff = 'Штат'
            and s2.business_dt in (select slot_end from hh_calendar where idx >= 0)
            and d.legal_position_rk::text <> coalesce(s2.legal_position_rk, '')) q
    join prod_v_sse_crossdata.legal_position_vacation x
        on x.legal_position_rk::text = q.rk
       and x.valid_from_dttm <= q.business_dt + interval '1 day' - interval '1 second'
       and x.valid_to_dttm   >= q.business_dt + interval '1 day' - interval '1 second'
    where x.deleted_flg = 0
    group by 1, 2
) w
    on w.mdm_employee_rk = s.mdm_employee_rk and w.business_dt = s.business_dt
where s.is_slot_row = 1 and s.hc = 1 and s.staff = 'Штат' and s.path_s <> ''
  and s.business_dt in (select slot_end from hh_calendar where idx >= 0)
distributed by (mdm_employee_rk);

analyze hh_vac;

drop table if exists hh_vac_cov;
create table hh_vac_cov as
select c.idx, c.slot_start, c.slot_end,
       count(v.mdm_employee_rk)                                          as staff_hc,
       count(v.days)                                                     as with_vac,
       round(count(v.days)::numeric / nullif(count(v.mdm_employee_rk), 0), 4) as cov
from hh_calendar c
cross join hh_param p
left join hh_vac v on v.business_dt = c.slot_end
where c.grain = 'm' and c.idx >= 0 and c.slot_end <= p.data_dt
group by 1, 2, 3
distributed replicated;

drop table if exists hh_vac_start;
create table hh_vac_start as
select coalesce(min(c.slot_start), date '5999-01-01') as vac_start_dt
from hh_vac_cov c
cross join hh_param p
where c.cov >= p.vac_min_cov
distributed replicated;

-- Справочно: покрытие по месяцам, месяц начала метрики и штатные с неосновными позициями в зонах.
select c.slot_end, c.staff_hc, c.with_vac, c.cov,
       (select count(*) from hh_vac x where x.business_dt = c.slot_end and x.xdays >= 28 and x.xdays < 56) as sec_yellow,
       (select count(*) from hh_vac x where x.business_dt = c.slot_end and x.xdays >= 56)                    as sec_red,
       case when c.slot_start >= s.vac_start_dt then 'в отчёте' else 'нет данных' end as vac_in_report
from hh_vac_cov c
cross join hh_vac_start s
order by c.idx;
```

## 09 HH · события по слотам — заменить код целиком (поставка 67: компоненты vxy, vxr — штатные с неосновными позициями в зоне)

```sql
-- ============================================================================
-- HH · события по слотам → hh_evt. Длинная таблица «вклад строки в компонент
-- метрики в слоте». Всё, что дальше, — суммы по ней.
--   запасы     hc, jun          сотрудник в списочной численности на конец слота — любого
--                               типа (hc из «строк источника»; «Активная» — фильтр hct);
--                               джун — по hh_seniority

--   найм       hire             в слоте дня найма в компанию (company_hire_flg)
--              hirea            в слоте прихода в активную численность (когорта 'a': найм с типом
--                               «Активная» + переход кандидата + внутренний перевод; HRBP 13.10)
--   hch        численность активной HQ на конец слота — знаменатель текучести (HRBP 13.10);
--              значения сравниваются без регистра и пробелов: в источнике покраска — «Hq» (14.10)
--   отпуска    vac, vacd, vy, vr (11.11, «Накопленный отпуск»): запас на конец слота, только штатные (hh_vac — основная
--              позиция, «HH · отпуска»): vacd — штатный в численности, vac — его остаток в сотых долях дня, vy / vr — в
--              жёлтой (28 ≤ дни < 56) / красной (≥ 56) зоне. Слоты раньше начала метрики (hh_vac_start) — нули: истории
--              отпусков там нет. Без записи об отпуске — в vacd, но не в зоне. vxy / vxr (12.11) — штатные, у кого сумма
--              по НЕОСНОВНЫМ позициям (hh_vac.xdays) в жёлтой / красной зоне: пометка у метрики, долю не меняют; в куб
--              атрибутов (11) не идут — в «Трансформерах» пометки нет.
--   когорты    r3d / r3n        месяцы (Дьяченко 19.10): новичок в слоте МЕСЯЦА НАЙМА — r3d, и работает ли
--              r6d / r6n        через 3 / 6 мес от найма — r3n; только те, у кого срок уже прошёл (найм + 3 (6)
--                               ≤ data_dt). Месяц, где срок прошёл не у всех, — дозревающий: чарт рисует его пунктиром.
--                               Недели — как было: в слоте созревания (найм + 13 / 26 нед). Суффикс a — база
--                               прихода в активную численность.
-- Слоты — с историей окон (idx < 0), окна применяются в следующем параграфе.
-- ============================================================================
drop table if exists hh_evt;
create table hh_evt as
select c.grain, c.idx, s.path_s, s.paint, s.it, s.stream, s.spec, s.staff, s.hct, s.seniority, s.a_grade, s.a_exp, s.a_gender, s.a_age, s.a_office, s.a_work, s.a_head, s.a_legal, s.a_macro, s.a_city, s.a_hrbp,
       1 as hc, coalesce(j.jun, 0) as jun, 0 as fire, 0 as rg, 0 as nrg, 0 as nr, 0 as nrd, 0 as hire, 0 as r3n, 0 as r3d, 0 as r6n, 0 as r6d, 0 as r3an, 0 as r3ad, 0 as r6an, 0 as r6ad,
       case when lower(trim(s.hct)) = lower(trim(p.active_type_value)) and lower(trim(s.paint)) = lower(trim(p.hq_paint_value)) then 1 else 0 end as hch, 0 as hirea,
       case when c.idx >= 0 and c.slot_end >= vs.vac_start_dt then coalesce(round(v.days * 100), 0)::bigint else 0 end as vac,
       case when c.idx >= 0 and c.slot_end >= vs.vac_start_dt and s.staff = 'Штат' then 1 else 0 end as vacd,
       case when c.idx >= 0 and c.slot_end >= vs.vac_start_dt and v.days >= 28 and v.days < 56 then 1 else 0 end as vy,
       case when c.idx >= 0 and c.slot_end >= vs.vac_start_dt and v.days >= 56 then 1 else 0 end as vr,
       case when c.idx >= 0 and c.slot_end >= vs.vac_start_dt and v.xdays >= 28 and v.xdays < 56 then 1 else 0 end as vxy,
       case when c.idx >= 0 and c.slot_end >= vs.vac_start_dt and v.xdays >= 56 then 1 else 0 end as vxr
from hh_src s
join hh_calendar c on c.slot_end = s.business_dt
left join hh_seniority j on j.seniority = s.seniority
cross join hh_param p
cross join hh_vac_start vs
left join hh_vac v on v.mdm_employee_rk = s.mdm_employee_rk and v.business_dt = s.business_dt
where s.is_slot_row = 1 and s.hc = 1 and s.path_s <> ''
union all
select c.grain, c.idx, f.path_s, f.paint, f.it, f.stream, f.spec, f.staff, f.hct, f.seniority, f.a_grade, f.a_exp, f.a_gender, f.a_age, f.a_office, f.a_work, f.a_head, f.a_legal, f.a_macro, f.a_city, f.a_hrbp,
       0 as hc, 0 as jun, 1 as fire, f.rg as rg, f.nrg as nrg, case when c.grain = 'm' then f.nr else 0 end as nr, case when c.grain = 'm' then f.nrd else 0 end as nrd, 0 as hire, 0 as r3n, 0 as r3d, 0 as r6n, 0 as r6d, 0 as r3an, 0 as r3ad, 0 as r6an, 0 as r6ad, 0 as hch, 0 as hirea, 0 as vac, 0 as vacd, 0 as vy, 0 as vr, 0 as vxy, 0 as vxr
from hh_fire f
join hh_calendar c on f.fire_dt between c.slot_start and c.slot_end
union all
select 'w'::text as grain, c.idx, f.path_s, f.paint, f.it, f.stream, f.spec, f.staff, f.hct, f.seniority, f.a_grade, f.a_exp, f.a_gender, f.a_age, f.a_office, f.a_work, f.a_head, f.a_legal, f.a_macro, f.a_city, f.a_hrbp,
       0 as hc, 0 as jun, 0 as fire, 0 as rg, 0 as nrg, f.nra as nr, f.nrda as nrd, 0 as hire, 0 as r3n, 0 as r3d, 0 as r6n, 0 as r6d, 0 as r3an, 0 as r3ad, 0 as r6an, 0 as r6ad, 0 as hch, 0 as hirea, 0 as vac, 0 as vacd, 0 as vy, 0 as vr, 0 as vxy, 0 as vxr
from hh_fire f
join hh_calendar c on c.grain = 'm' and f.fire_dt between c.slot_start and c.slot_end
where f.nrda = 1
union all
select c.grain, c.idx, h.path_s, h.paint, h.it, h.stream, h.spec, h.staff, h.hct, h.seniority, h.a_grade, h.a_exp, h.a_gender, h.a_age, h.a_office, h.a_work, h.a_head, h.a_legal, h.a_macro, h.a_city, h.a_hrbp,
       0 as hc, 0 as jun, 0 as fire, 0 as rg, 0 as nrg, 0 as nr, 0 as nrd, case when h.base = 'c' then 1 else 0 end as hire, 0 as r3n, 0 as r3d, 0 as r6n, 0 as r6d, 0 as r3an, 0 as r3ad, 0 as r6an, 0 as r6ad,

from hh_cohort h
join hh_calendar c on h.hire_dt between c.slot_start and c.slot_end
union all
select c.grain, c.idx, h.path_s, h.paint, h.it, h.stream, h.spec, h.staff, h.hct, h.seniority, h.a_grade, h.a_exp, h.a_gender, h.a_age, h.a_office, h.a_work, h.a_head, h.a_legal, h.a_macro, h.a_city, h.a_hrbp,
       0 as hc, 0 as jun, 0 as fire, 0 as rg, 0 as nrg, 0 as nr, 0 as nrd, 0 as hire, case when h.base = 'c' and h.hz = 3 and h.ret = 1 then 1 else 0 end as r3n, case when h.base = 'c' and h.hz = 3 then 1 else 0 end as r3d, case when h.base = 'c' and h.hz = 6 and h.ret = 1 then 1 else 0 end as r6n, case when h.base = 'c' and h.hz = 6 then 1 else 0 end as r6d, case when h.base = 'a' and h.hz = 3 and h.ret = 1 then 1 else 0 end as r3an, case when h.base = 'a' and h.hz = 3 then 1 else 0 end as r3ad, case when h.base = 'a' and h.hz = 6 and h.ret = 1 then 1 else 0 end as r6an, case when h.base = 'a' and h.hz = 6 then 1 else 0 end as r6ad, 0 as hch, 0 as hirea, 0 as vac, 0 as vacd, 0 as vy, 0 as vr, 0 as vxy, 0 as vxr

    select q.*,
           case when q.fire_dt is null or q.fire_dt >= q.md then 1 else 0 end as ret
    from (
        select h.*, z.grain as zgrain, z.hz,
               (case when z.mons is not null then h.hire_dt + z.mons * interval '1 month'
                     else h.hire_dt + z.dys * interval '1 day' end)::date as md
        from hh_cohort h
        cross join (values ('m'::text, 3, 3, null::int), ('m'::text, 6, 6, null::int),
                           ('w'::text, 3, null::int, 91), ('w'::text, 6, null::int, 182)) z(grain, hz, mons, dys)
    ) q
) h
join hh_calendar c on c.grain = h.zgrain
    and (case when h.zgrain = 'm' then h.hire_dt else h.md end) between c.slot_start and c.slot_end
cross join hh_param p
where h.md <= p.data_dt
distributed by (path_s);

analyze hh_evt;
```

## 10 HH · куб по разрезам — заменить код целиком (поставка 67: массивы vxy, vxr в кубе и базе)

```sql
-- ============================================================================
-- HH · куб по разрезам → hrbp_hub_cube. Ключ строки — путь юнита × 6 разрезов
-- численности; значения — массивы по слотам: m_* 24 месяца, w_* 24 недели
-- (позиция = idx из hrbp_hub_calendar). Проценты не храним — только числитель и
-- знаменатель, поэтому любую строку можно складывать с любой другой.
--
-- ОКНА (меняются здесь и только здесь):
--   rg, nrg (regrettable / non-regrettable увольнения) — 12 мес / 52 нед;
--   hcw — знаменатель текучести, всегда по АКТИВНОЙ численности HQ, какие бы разрезы ни были выбраны
--   (HRBP 13.10, Дьяченко): месяцы — численность активной HQ на конец месяца (hch, без окна: из неё чарт
--   считает текучесть с начала года), недели — Σ за 52 нед (как раньше; в отчёте не показываются);
--   nr (незаполненные причины) — с начала года (окно wy — с января того же года): увольнения с 1 января, у которых
--   прошло 30+ дней, а причины нет (HRBP 13.10);
--   nrd — знаменатель nr тем же окном: все увольнения с 1 января, у которых прошло 30+ дней (Дьяченко 16.10);
--   w_nr / w_nrd (02.11, поставка 54) — НЕ недели: те же месяцы и то же окно wy, но все уволенные, и свежие (у кого 30 дней
--   ещё не прошли; «HH · события по слотам» кладёт их в grain = 'w' с idx месяца) — по ним чарт рисует дозревающий месяц;
--   hirea — найм в активную численность (без окна): найм в компанию с типом «Активная» + переходы
--   кандидатов + внутренние переводы (candidate_transfer_flg, internal_transfer_flg; HRBP 13.10);
--   r3*/r6* (когорта найма) — месяцы (Дьяченко 19.10, было — когорта с начала года по слоту созревания): БЕЗ окна,
--          в слоте МЕСЯЦА НАЙМА. Знаменатель — все нанятые в этом месяце, числитель — те из них, кто работает
--          через 3 (6) мес от найма — только те, у кого срок уже прошёл (найм + 3 (6) ≤ data_dt). Месяц, где срок
--          прошёл у всех (конец месяца «найм + 3 (6)» ≤ data_dt), — созревший; остальные — дозревающие (в чарте —
--          пунктиром). Средняя с начала года (Σ числителей / Σ знаменателей созревших месяцев с января) — в чарте. Недели — как было: 39 и 26 нед по слоту созревания (в
--          отчёте не показываются).
--   hc, jun, hire, fire — без окна (запас на конец слота и поток слота).
--   rgs, nrgs — те же regrettable / non-regrettable увольнения БЕЗ окна (поток слота): по ним чарт
--               считает текучесть нарастающим итогом с начала года (владелец 12.10).
-- Окно считается на каждой строке куба: сумма окон по строкам = окно суммы.
-- ============================================================================

                   coalesce(l.r6ad, 0) as r6ad,
                   coalesce(l.hch, 0) as hch,
                   coalesce(l.hirea, 0) as hirea,
                   coalesce(l.vac, 0) as vac,

    left join hh_cube_long l

window w12 as (partition by x.kid, x.grain order by x.idx rows between 11 preceding and current row),
       wy as (partition by x.kid, x.grain, floor(x.idx / 12.0) order by x.idx rows between unbounded preceding and current row),

       w39 as (partition by x.kid, x.grain order by x.idx rows between 38 preceding and current row),
       w26 as (partition by x.kid, x.grain order by x.idx rows between 25 preceding and current row)
distributed by (kid);
```

```sql
-- 1. Дубли источника на днях, которые читает ноут: одна строка на (сотрудник, день).
select 1 / (case when count(*) > 0 then 0 else 1 end) as source_duplicates_check
from (
    select e.mdm_employee_rk, e.business_dt
    from prod_v_emart.mdm_employee_structure_d e
    join (select distinct c.slot_end as d from hh_calendar c cross join hh_param p where c.slot_end <= p.data_dt) sd
        on sd.d = e.business_dt
    group by 1, 2
    having count(*) > 1
) t;

-- 2. Короткие id узлов уникальны (первые 12 знаков md5 от rk).
select 1 / (case when count(distinct rk) = count(distinct id) and count(*) = count(distinct id) then 1 else 0 end) as unit_id_check
from hrbp_hub_unit;

-- 3. Все массивы куба — ровно 24 позиции.
select 1 / (case when count(*) > 0 then 0 else 1 end) as cube_array_length_check
from hrbp_hub_cube
where array_length(m_hc, 1) <> 24 or array_length(w_hc, 1) <> 24
   or array_length(m_hcw, 1) <> 24 or array_length(w_r6ad, 1) <> 24
   or array_length(m_vr, 1) <> 24 or array_length(w_vr, 1) <> 24
   or array_length(m_vxr, 1) <> 24 or array_length(w_vxr, 1) <> 24;
select 1 / (case when count(*) > 0 then 0 else 1 end) as attr_array_length_check
from hrbp_hub_attr
where array_length(m_hc, 1) <> 24 or array_length(m_r6ad, 1) <> 24 or array_length(m_vr, 1) <> 24;
select 1 / (case when count(*) > 0 then 0 else 1 end) as attr_top_array_length_check
from hrbp_hub_attr_top
where array_length(m_hc, 1) <> 24 or array_length(m_r6ad, 1) <> 24 or array_length(m_vr, 1) <> 24;

-- 3б. Свёртка атрибутов = листовой куб: свёртки верхних узлов дерева (pid = '') вместе —
--     это весь куб юридической численности (свёртка — «без разрезов», 13.10); по каждому атрибуту сходятся
--     численность, увольнения и накопленный отпуск (штатные и их остаток, 11.11) за все месяцы.
select 1 / (case when count(*) > 0 then 0 else 1 end) as attr_top_equals_leaf_check
from (
    select coalesce(l.attr_k, t.attr_k) as attr_k
    from (select x.attr_k, sum(x.hc) as hc, sum(x.fire) as fire, sum(x.vacd) as vacd, sum(x.vac) as vac
          from (select attr_k, unnest(m_hc) as hc, unnest(m_fire) as fire, unnest(m_vacd) as vacd, unnest(m_vac) as vac
                from hrbp_hub_attr a cross join hh_param pr where a.hct <> pr.nonlegal_type) x
          group by 1) l
    full join (select x.attr_k, sum(x.hc) as hc, sum(x.fire) as fire, sum(x.vacd) as vacd, sum(x.vac) as vac
               from (select a.attr_k, unnest(a.m_hc) as hc, unnest(a.m_fire) as fire, unnest(a.m_vacd) as vacd, unnest(a.m_vac) as vac
                     from hrbp_hub_attr_top a
                     join hrbp_hub_unit u on u.id = a.unit_id and u.pid = '') x
               group by 1) t
        on t.attr_k = l.attr_k
    where coalesce(l.hc, -1) <> coalesce(t.hc, -1) or coalesce(l.fire, -1) <> coalesce(t.fire, -1)
       or coalesce(l.vacd, -1) <> coalesce(t.vacd, -1) or coalesce(l.vac, -1) <> coalesce(t.vac, -1)
) d;

-- 4. Куб сходится с источником: численность на конец последнего закрытого месяца.
select 1 / (case when c.hc = s.hc then 1 else 0 end) as cube_headcount_check, c.hc as cube_hc, s.hc as source_hc
from (
    select sum(k.m_hc[l.pos]) as hc
    from hrbp_hub_cube k
    cross join (select c.idx + 1 as pos
                from hrbp_hub_calendar c
                join hh_param p on c.grain = 'm' and c.slot_end = p.last_month_end) l
) c
cross join (
    select count(*) as hc
    from hh_src s join hh_param p on s.business_dt = p.last_month_end
    where s.is_slot_row = 1 and s.hc = 1 and s.path_s <> ''
) s;

-- 4б. База сходится с кубом: вся компания без пути = сумма куба (месяцы и недели,
--     численность и знаменатель текучести) и массивы по 24 позиции.
select 1 / (case when b.m_hc = c.m_hc and b.w_hc = c.w_hc and b.m_hcw = c.m_hcw and b.bad = 0 then 1 else 0 end) as base_equals_cube_check,
       b.m_hc as base_hc, c.m_hc as cube_hc
from (
    select sum(k.m_hc[l.pos]) as m_hc, sum(k.w_hc[24]) as w_hc, sum(k.m_hcw[l.pos]) as m_hcw,
           sum(case when array_length(k.m_hc, 1) <> 24 or array_length(k.w_r6ad, 1) <> 24 then 1 else 0 end) as bad
    from hrbp_hub_base k
    cross join (select c.idx + 1 as pos
                from hrbp_hub_calendar c
                join hh_param p on c.grain = 'm' and c.slot_end = p.last_month_end) l
) b
cross join (
    select sum(k.m_hc[l.pos]) as m_hc, sum(k.w_hc[24]) as w_hc, sum(k.m_hcw[l.pos]) as m_hcw
    from hrbp_hub_cube k
    cross join (select c.idx + 1 as pos
                from hrbp_hub_calendar c
                join hh_param p on c.grain = 'm' and c.slot_end = p.last_month_end) l
) c;

-- 4г. Уволенные без причины (поставка 56) сходятся с метрикой: всей компании с начала года — с прошедшими 30 днями
--     столько же, сколько m_nr базы в последнем месяце, всех (и свежих) — сколько w_nr.
select 1 / (case when p.r = b.r and p.a = b.a then 1 else 0 end) as nr_people_check,
       p.r as people_ripe, b.r as base_nr, p.a as people_all, b.a as base_nr_all
from (select coalesce(sum(ripe), 0) as r, count(*) as a from hrbp_hub_nr_people) p
cross join (
    select coalesce(sum(k.m_nr[l.pos]), 0) as r, coalesce(sum(k.w_nr[l.pos]), 0) as a
    from hrbp_hub_base k
    cross join (select c.idx + 1 as pos
                from hrbp_hub_calendar c
                join hh_param p on c.grain = 'm' and c.slot_end = p.last_month_end) l
) b;

-- 4д. Накопленный отпуск (11.11) сходится с «HH · отпуска»: на конец последнего месяца штатные в численности (vacd),
--     их остаток в сотых долях дня (vac) и люди в жёлтой / красной зоне (vy / vr) — в кубе, в базе и в hh_vac одно и
--     то же; если метрика ещё не началась (покрытие ниже порога) — везде нули. С 12.11 — и штатные, у кого сумма по
--     неосновным позициям в жёлтой / красной зоне (vxy / vxr, sec_*).
select 1 / (case when c.d = s.d and c.v = s.v and c.y = s.y and c.r = s.r and c.xy = s.xy and c.xr = s.xr
                  and b.d = s.d and b.v = s.v and b.y = s.y and b.r = s.r and b.xy = s.xy and b.xr = s.xr
             then 1 else 0 end) as vac_check,
       s.d as src_staff, c.d as cube_staff, s.v as src_vac, c.v as cube_vac, s.y as src_yellow, c.y as cube_yellow,
       s.r as src_red, c.r as cube_red, s.xy as sec_yellow, c.xy as cube_sec_yellow, s.xr as sec_red, c.xr as cube_sec_red
from (
    select coalesce(sum(k.m_vacd[l.pos]), 0) as d, coalesce(sum(k.m_vac[l.pos]), 0) as v,
           coalesce(sum(k.m_vy[l.pos]), 0) as y, coalesce(sum(k.m_vr[l.pos]), 0) as r,
           coalesce(sum(k.m_vxy[l.pos]), 0) as xy, coalesce(sum(k.m_vxr[l.pos]), 0) as xr
    from hrbp_hub_cube k
    cross join (select c.idx + 1 as pos
                from hrbp_hub_calendar c
                join hh_param p on c.grain = 'm' and c.slot_end = p.last_month_end) l
) c
cross join (
    select coalesce(sum(k.m_vacd[l.pos]), 0) as d, coalesce(sum(k.m_vac[l.pos]), 0) as v,
           coalesce(sum(k.m_vy[l.pos]), 0) as y, coalesce(sum(k.m_vr[l.pos]), 0) as r,
           coalesce(sum(k.m_vxy[l.pos]), 0) as xy, coalesce(sum(k.m_vxr[l.pos]), 0) as xr
    from hrbp_hub_base k
    cross join (select c.idx + 1 as pos
                from hrbp_hub_calendar c
                join hh_param p on c.grain = 'm' and c.slot_end = p.last_month_end) l
) b
cross join (
    select coalesce(sum(o.ison), 0) as d,
           coalesce(sum(o.ison * round(v.days * 100)), 0) as v,
           coalesce(sum(o.ison * case when v.days >= 28 and v.days < 56 then 1 else 0 end), 0) as y,
           coalesce(sum(o.ison * case when v.days >= 56 then 1 else 0 end), 0) as r,
           coalesce(sum(o.ison * case when v.xdays >= 28 and v.xdays < 56 then 1 else 0 end), 0) as xy,
           coalesce(sum(o.ison * case when v.xdays >= 56 then 1 else 0 end), 0) as xr
    from hh_vac v
    join hh_param p on v.business_dt = p.last_month_end
    cross join (select case when p.last_month_end >= s.vac_start_dt then 1 else 0 end as ison
                from hh_param p cross join hh_vac_start s) o
) s;

-- 4в. Глубина пути: компания + не больше 10 уровней (lvl3…lvl12).
select 1 / (case when max(array_length(path, 1)) <= 11 then 1 else 0 end) as path_depth_check,
       max(array_length(path, 1)) as max_path_len
from hrbp_hub_cube;

-- 5. Каждый узел из путей куба есть в справочнике.
select 1 / (case when count(*) > 0 then 0 else 1 end) as cube_units_in_dictionary_check
from (select distinct unnest(path) as id from hrbp_hub_cube) x
left join hrbp_hub_unit u on u.id = x.id
where u.id is null;

-- 6. Доступ не пуст: хотя бы один HRBP с зоной.
select 1 / (case when count(*) > 0 then 1 else 0 end) as access_not_empty_check
from hrbp_hub_access where role in ('hrbp', 'super');

-- Диагностика: строки без юнита (в отчёт не попадают), увольнения без разметки
-- regrettable, незаполненные причины; правила KPI, не попавшие в отчёт.
select (select count(*) from hh_src where path_s = '')                     as rows_without_unit,
       (select count(*) from hh_fire)                                        as fires,
       (select count(*) from hh_fire where rg = 0 and nrg = 0)               as fires_without_regret_marking,
       (select count(*) from hh_fire where nr = 1)                           as fires_without_reason,
       (select count(*) from hrbp_hub_cube)                                  as cube_rows,
       (select count(*) from hrbp_hub_attr)                                  as attr_rows,
       (select count(*) from hrbp_hub_base)                                  as base_rows,
       (select count(*) from hrbp_hub_unit where is_current = 1)             as units_now,
       (select count(distinct login) from hrbp_hub_access)                   as logins;
select r.rule_id, r.unit_rk, r.metric_id,
       case when u.id is null then 'нет такого юнита' else 'неизвестная метрика или пустая цель' end as problem
from hh_kpi_src r
left join hrbp_hub_unit u on u.rk = trim(r.unit_rk)
where u.id is null
   or trim(r.metric_id) not in ('retention_new_3', 'retention_new_6', 'regret', 'exit_reasons', 'jun_team', 'vac_yellow', 'vac_red')
   or r.target is null;

-- Диагностика отпусков (11.11): с какого месяца метрика в отчёте (vac_start_dt; 5999-01-01 — ни один месяц не набрал
-- порог покрытия), покрытие штатных записями об отпуске в последнем месяце и сколько месяцев после начала ниже порога
-- (их отчёт показывает — проверить источник; помесячно — таблица hh_vac_cov).
select s.vac_start_dt,
       (select c.cov from hh_vac_cov c where c.slot_end = p.last_month_end)                          as vac_cov_now,
       p.vac_min_cov,
       (select count(*) from hh_vac_cov c where c.slot_start >= s.vac_start_dt and c.cov < p.vac_min_cov) as vac_low_months
from hh_vac_start s
cross join hh_param p;
```
