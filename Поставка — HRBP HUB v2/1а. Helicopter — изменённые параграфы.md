```sql
-- ============================================================================
-- HH · строки источника. Один проход по mdm_employee_structure_d — дальше всё
-- считается отсюда.
--   is_slot_row  строка на конец слота (месяц / неделя, вкл. историю окон) или на
--                срез data_dt: запасы (численность, джуны), справочник, зоны HRBP;
--   is_fire_row  строка увольнения из компании — company_fire_flg = 1 (HRBP 13.10: отток всегда по нему);
--   is_hire_row  строка найма в компанию — company_hire_flg = 1;
--   is_ahire_row строка прихода в активную численность (HRBP 13.10, Дьяченко): найм в компанию с типом
--                «Активная» + переход кандидата (candidate_transfer_flg) + внутренний перевод
--                (internal_transfer_flg) — найм и когорта закрепляемости при фильтре «Активная».
--   hct          тип численности; у строк вне юридической численности (legal_employee_flg <> 1) —
--                значение hh_param.nonlegal_type: без выбранного типа отчёт их не берёт (HRBP 13.10).
--   hc           в списочной численности на эту дату — ЛЮБОГО типа (active_type_nm):
--                нанят (company_hire_dt <= дата) и не уволен раньше (company_fire_dt пусто
--                или >= дата) и не уволен в этот день (company_fire_flg = 0: Дьяченко 16.10 — в численности
--                всегда не уволенные; до 16.10 уволенный в день среза ещё был в ней). Активная численность —
--                фильтр «Тип численности» = «Активная»
--                (владелец 12.10: все метрики — и по активной, и по всем типам; до 12.10
--                hc = active_employee_flg, и у прочих типов численность была 0).
--   hc_act       active_employee_flg — для «Диагностики источников» (сверка с hct).
-- Атрибуты строки — на ЕЁ дату: запасы берут разрезы на конец периода, события —
-- на дату события. Путь юнита — mapped-цепочка, как в ультраширокой: lvl1
-- (компания) и lvl3…lvl12. lvl2 (группировка юрлиц над блоками) пропускается —
-- блоки встают прямо под компанию; глубже lvl12 отчёт не спускается: сотрудник
-- lvl13 считается прямо в своём юните 12-го уровня, сотрудник lvl1/lvl2 — прямо
-- в компании. id узла = первые 12 знаков md5(rk): короткий, стабильный между
-- пересборками.
-- Руководитель юнита (13.10, логин под именем юнита в фильтре): is_head — management_head_flg, login —
-- ad_login, head_login — management_head_ad_login (руководитель юнита сотрудника, как head_login
-- ультраширокой), deep13 — сотрудник глубже 12-го уровня (его руководитель — не руководитель юнита отчёта).
-- Разрезы численности (как в filter-fields.md):
--   paint  = emp_specialization_oper_code      it    = emp_specialization_it_code
--   stream = emp_stream_desc                   spec  = emp_specialization_desc
--   staff  = «Штат» для employment_relation_type_desc = 'Штатный сотрудник', иначе «Не штат»
--   hct    = active_type_nm как есть
-- Пустое значение разреза = '-'. Атрибуты a_* — оси трансформеров.
-- ============================================================================
drop table if exists hh_src_raw;
create table hh_src_raw as
select
    e.business_dt::date                                                        as business_dt,
    e.mdm_employee_rk,
    case when sd.d is not null then 1 else 0 end                                as is_slot_row,
    case when coalesce(e.company_fire_flg, 0) = 1 then 1 else 0 end            as is_fire_row,
    case when coalesce(e.company_hire_flg, 0) = 1 then 1 else 0 end            as is_hire_row,
    case when (coalesce(e.company_hire_flg, 0) = 1 and lower(trim(e.active_type_nm)) = lower(trim(pa.active_type_value)))
           or coalesce(e.candidate_transfer_flg, 0) = 1
           or coalesce(e.internal_transfer_flg, 0) = 1 then 1 else 0 end      as is_ahire_row,
    case when coalesce(e.company_hire_dt, e.business_dt) <= e.business_dt
          and (e.company_fire_dt is null or e.company_fire_dt >= e.business_dt)
          and coalesce(e.company_fire_flg, 0) = 0
         then 1 else 0 end                                                     as hc,
    coalesce(e.active_employee_flg, 0)::int                                    as hc_act,
    e.company_hire_dt::date                                                    as company_hire_dt,
    e.active_hire_dt::date                                                     as active_hire_dt,
    e.legal_fire_dt::date                                                      as legal_fire_dt,
    e.legal_position_rk::text                                                  as legal_position_rk,
    concat_ws('/',
        substr(md5(nullif(nullif(trim(e.lvl1_mapped_management_unit_rk::text), ''), '-')), 1, 12),
        substr(md5(nullif(nullif(trim(e.lvl3_mapped_management_unit_rk::text), ''), '-')), 1, 12),
        substr(md5(nullif(nullif(trim(e.lvl4_mapped_management_unit_rk::text), ''), '-')), 1, 12),
        substr(md5(nullif(nullif(trim(e.lvl5_mapped_management_unit_rk::text), ''), '-')), 1, 12),
        substr(md5(nullif(nullif(trim(e.lvl6_mapped_management_unit_rk::text), ''), '-')), 1, 12),
        substr(md5(nullif(nullif(trim(e.lvl7_mapped_management_unit_rk::text), ''), '-')), 1, 12),
        substr(md5(nullif(nullif(trim(e.lvl8_mapped_management_unit_rk::text), ''), '-')), 1, 12),
        substr(md5(nullif(nullif(trim(e.lvl9_mapped_management_unit_rk::text), ''), '-')), 1, 12),
        substr(md5(nullif(nullif(trim(e.lvl10_mapped_management_unit_rk::text), ''), '-')), 1, 12),
        substr(md5(nullif(nullif(trim(e.lvl11_mapped_management_unit_rk::text), ''), '-')), 1, 12),
        substr(md5(nullif(nullif(trim(e.lvl12_mapped_management_unit_rk::text), ''), '-')), 1, 12)
    )                                                                          as path_s,
    coalesce(nullif(trim(e.emp_specialization_oper_code::text), ''), '-')      as paint,
    coalesce(nullif(trim(e.emp_specialization_it_code::text), ''), '-')        as it,
    coalesce(nullif(trim(e.emp_stream_desc::text), ''), '-')                   as stream,
    coalesce(nullif(trim(e.emp_specialization_desc::text), ''), '-')           as spec,
    case when e.employment_relation_type_desc = 'Штатный сотрудник'
         then 'Штат' else 'Не штат' end                                        as staff,
    case when coalesce(e.legal_employee_flg, 0) <> 1 then pa.nonlegal_type
         else coalesce(nullif(trim(e.active_type_nm::text), ''), '-') end      as hct,
    coalesce(nullif(trim(e.seniority::text), ''), '-')                         as seniority,
    coalesce(nullif(trim(e.grade::text), ''), '-')                             as a_grade,
    coalesce(nullif(trim(e.experience_group_nm::text), ''), '-')               as a_exp,
    coalesce(nullif(trim(e.gender_desc::text), ''), '-')                       as a_gender,
    case when e.birth_dt is null then '-'
         when extract(year from age(e.business_dt::date, e.birth_dt::date)) < 25 then 'до 25'
         when extract(year from age(e.business_dt::date, e.birth_dt::date)) < 35 then '25–34'
         when extract(year from age(e.business_dt::date, e.birth_dt::date)) < 45 then '35–44'
         else '45 и старше' end                                                as a_age,
    coalesce(nullif(trim(e.office_desc::text), ''), '-')                       as a_office,
    coalesce(nullif(trim(e.employee_contract_type_desc::text), ''), '-')       as a_work,
    case when coalesce(e.management_head_flg, 0) = 1
         then 'Руководитель' else 'Не руководитель' end                        as a_head,
    coalesce(nullif(trim(e.lvl1_legal_unit_nm::text), ''), '-')                as a_legal,
    coalesce(nullif(trim(r.macroregion_nm::text), ''), '-')                    as a_macro,
    coalesce(nullif(trim(r.city_nm::text), ''), '-')                           as a_city,
    r.valid_from_dttm                                                          as res_from,
    e.hrbp_mdm_employee_rk                                                     as hrbp_rk,
    lower(trim(e.hrbp_ad_login::text))                                         as hrbp_login,
    trim(coalesce(e.hrbp_last_nm::text, '') || ' ' || coalesce(e.hrbp_first_nm::text, '')) as hrbp_nm,
    coalesce(e.management_head_flg, 0)::int                                    as is_head,
    nullif(lower(trim(e.ad_login::text)), '')                                  as login,
    nullif(lower(trim(e.management_head_ad_login::text)), '')                  as head_login,
    case when nullif(nullif(trim(e.lvl13_mapped_management_unit_rk::text), ''), '-') is not null
         then 1 else 0 end                                                     as deep13
from prod_v_emart.mdm_employee_structure_d e
cross join (select min(slot_start) - interval '7 month' as d0 from hh_calendar) lim
cross join hh_param pa
left join (select c.slot_end as d from hh_calendar c cross join hh_param p where c.slot_end <= p.data_dt
           union
           select data_dt from hh_param) sd
    on sd.d = e.business_dt
left join prod_v_sse_crossdata.mdm_employee_residence r
    on r.mdm_employee_rk = e.mdm_employee_rk
   and e.business_dt between r.valid_from_dttm and r.valid_to_dttm
   and r.deleted_flg = 0
where e.business_dt >= lim.d0
  and (sd.d is not null
       or coalesce(e.company_fire_flg, 0) = 1
       or coalesce(e.company_hire_flg, 0) = 1
       or coalesce(e.candidate_transfer_flg, 0) = 1
       or coalesce(e.internal_transfer_flg, 0) = 1)
distributed by (mdm_employee_rk);

-- Одна строка на (сотрудник, день): пересечения периодов проживания дают дубли —
-- берём самый свежий. Дубли самого источника проверяет параграф «HH · проверки».
drop table if exists hh_src;
create table hh_src as
select distinct on (mdm_employee_rk, business_dt) *
from hh_src_raw
order by mdm_employee_rk, business_dt, res_from desc
distributed by (mdm_employee_rk);

-- Джун — по seniority и шаблонам из hh_param_junior (сопоставление по ILIKE, не точное).
drop table if exists hh_seniority;
create table hh_seniority as
select s.seniority,
       max(case when s.seniority ilike j.pattern then 1 else 0 end) as jun
from (select distinct seniority from hh_src) s
cross join hh_param_junior j
group by s.seniority
distributed replicated;

analyze hh_src;
```

## 07 HH · увольнения — заменить код целиком (новый признак nrd — знаменатель незаполненных причин: все уволенные, по которым прошло 30 дней)

```sql
-- ============================================================================
-- HH · увольнения → hh_fire. Событие = строка дня увольнения из компании
-- (business_dt = company_fire_dt); юнит и разрезы — на эту дату.
--   rg / nrg — regrettable / non-regrettable: джойн вспомогательной витрины
--              usr_cross_data.regrettable_n_non_regrettable_base по сотруднику и
--              дате (как в ультраширокой); в базе этих флагов нет;
--   nr       — причина НЕ заполнена: legal_fire_dt не пустая, в
--              legal_position_dismissal_reason (legal_position_rk + fire_dt =
--              legal_fire_dt) нет причины — и с увольнения прошло nr_grace_days.
--              coalesce(dismissal_reason_desc, 'Причина отсутствует') = отсутствует.
--   nrd      — знаменатель незаполненных причин (Дьяченко 16.10: «не численность юнита, а все уволенные»):
--              то же, что nr, но с причиной или без — legal_fire_dt не пустая и прошло nr_grace_days.
-- Дни увольнения берутся с запасом в 7 месяцев до окна: по ним когорты найма
-- узнают, дожил ли новичок до 3 и 6 месяцев.
-- ============================================================================
drop table if exists hh_fire;
create table hh_fire as
select s.mdm_employee_rk, s.company_hire_dt, s.business_dt as fire_dt,
       s.path_s, s.paint, s.it, s.stream, s.spec, s.staff, s.hct,
       s.seniority, s.a_grade, s.a_exp, s.a_gender, s.a_age, s.a_office, s.a_work, s.a_head,
       s.a_legal, s.a_macro, s.a_city,
       coalesce(rt.rg, 0)  as rg,
       coalesce(rt.nrg, 0) as nrg,
       case when s.legal_fire_dt is not null
             and coalesce(fr.has_reason, 0) = 0
             and s.business_dt <= p.data_dt - p.nr_grace_days
            then 1 else 0 end as nr,
       case when s.legal_fire_dt is not null
             and s.business_dt <= p.data_dt - p.nr_grace_days
            then 1 else 0 end as nrd
from hh_src s
cross join hh_param p
left join (
    select mdm_employee_rk, business_dt::date as business_dt,
           max(case when regrettable_fire_flg = 1 then 1 else 0 end)     as rg,
           max(case when non_regrettable_fire_flg = 1 then 1 else 0 end) as nrg
    from usr_cross_data.regrettable_n_non_regrettable_base
    where business_dt >= (select min(slot_start) - interval '7 month' from hh_calendar)
    group by 1, 2
) rt on rt.mdm_employee_rk = s.mdm_employee_rk and rt.business_dt = s.business_dt
left join (
    select legal_position_rk::text as legal_position_rk, fire_dt::date as fire_dt,
           max(case when nullif(trim(dismissal_reason_desc::text), '') is not null
                     and dismissal_reason_desc <> 'Причина отсутствует' then 1 else 0 end) as has_reason
    from prod_v_hrmart.legal_position_dismissal_reason
    group by 1, 2
) fr on fr.legal_position_rk = s.legal_position_rk and fr.fire_dt = s.legal_fire_dt
where s.is_fire_row = 1
  and s.path_s <> ''
distributed by (mdm_employee_rk);
```

## 09 HH · события по слотам — заменить код целиком (поставка 44: закрепляемость — и дозревающие месяцы найма; поставка 43: когорта в месяце найма; поставка 39: nrd в событиях)

```sql
-- ============================================================================
-- HH · события по слотам → hh_evt. Длинная таблица «вклад строки в компонент
-- метрики в слоте». Всё, что дальше, — суммы по ней.
--   запасы     hc, jun          сотрудник в списочной численности на конец слота — любого
--                               типа (hc из «строк источника»; «Активная» — фильтр hct);
--                               джун — по hh_seniority
--   поток      fire, rg, nrg, nr, nrd в слоте дня увольнения
--   найм       hire             в слоте дня найма в компанию (company_hire_flg)
--              hirea            в слоте прихода в активную численность (когорта 'a': найм с типом
--                               «Активная» + переход кандидата + внутренний перевод; HRBP 13.10)
--   hch        численность активной HQ на конец слота — знаменатель текучести (HRBP 13.10);
--              значения сравниваются без регистра и пробелов: в источнике покраска — «Hq» (14.10)
--   когорты    r3d / r3n        месяцы (Дьяченко 19.10): новичок в слоте МЕСЯЦА НАЙМА — r3d, и работает ли
--              r6d / r6n        через 3 / 6 мес от найма — r3n; только те, у кого срок уже прошёл (найм + 3 (6)
--                               ≤ data_dt). Месяц, где срок прошёл не у всех, — дозревающий: чарт рисует его пунктиром.
--                               Недели — как было: в слоте созревания (найм + 13 / 26 нед). Суффикс a — база
--                               прихода в активную численность.
-- Слоты — с историей окон (idx < 0), окна применяются в следующем параграфе.
-- ============================================================================
drop table if exists hh_evt;
create table hh_evt as
select c.grain, c.idx, s.path_s, s.paint, s.it, s.stream, s.spec, s.staff, s.hct, s.seniority, s.a_grade, s.a_exp, s.a_gender, s.a_age, s.a_office, s.a_work, s.a_head, s.a_legal, s.a_macro, s.a_city,
       1 as hc, coalesce(j.jun, 0) as jun, 0 as fire, 0 as rg, 0 as nrg, 0 as nr, 0 as nrd, 0 as hire, 0 as r3n, 0 as r3d, 0 as r6n, 0 as r6d, 0 as r3an, 0 as r3ad, 0 as r6an, 0 as r6ad,
       case when lower(trim(s.hct)) = lower(trim(p.active_type_value)) and lower(trim(s.paint)) = lower(trim(p.hq_paint_value)) then 1 else 0 end as hch, 0 as hirea
from hh_src s
join hh_calendar c on c.slot_end = s.business_dt
left join hh_seniority j on j.seniority = s.seniority
cross join hh_param p
where s.is_slot_row = 1 and s.hc = 1 and s.path_s <> ''
union all
select c.grain, c.idx, f.path_s, f.paint, f.it, f.stream, f.spec, f.staff, f.hct, f.seniority, f.a_grade, f.a_exp, f.a_gender, f.a_age, f.a_office, f.a_work, f.a_head, f.a_legal, f.a_macro, f.a_city,
       0 as hc, 0 as jun, 1 as fire, f.rg as rg, f.nrg as nrg, f.nr as nr, f.nrd as nrd, 0 as hire, 0 as r3n, 0 as r3d, 0 as r6n, 0 as r6d, 0 as r3an, 0 as r3ad, 0 as r6an, 0 as r6ad, 0 as hch, 0 as hirea
from hh_fire f
join hh_calendar c on f.fire_dt between c.slot_start and c.slot_end
union all
select c.grain, c.idx, h.path_s, h.paint, h.it, h.stream, h.spec, h.staff, h.hct, h.seniority, h.a_grade, h.a_exp, h.a_gender, h.a_age, h.a_office, h.a_work, h.a_head, h.a_legal, h.a_macro, h.a_city,
       0 as hc, 0 as jun, 0 as fire, 0 as rg, 0 as nrg, 0 as nr, 0 as nrd, case when h.base = 'c' then 1 else 0 end as hire, 0 as r3n, 0 as r3d, 0 as r6n, 0 as r6d, 0 as r3an, 0 as r3ad, 0 as r6an, 0 as r6ad,
       0 as hch, case when h.base = 'a' then 1 else 0 end as hirea
from hh_cohort h
join hh_calendar c on h.hire_dt between c.slot_start and c.slot_end
union all
select c.grain, c.idx, h.path_s, h.paint, h.it, h.stream, h.spec, h.staff, h.hct, h.seniority, h.a_grade, h.a_exp, h.a_gender, h.a_age, h.a_office, h.a_work, h.a_head, h.a_legal, h.a_macro, h.a_city,
       0 as hc, 0 as jun, 0 as fire, 0 as rg, 0 as nrg, 0 as nr, 0 as nrd, 0 as hire, case when h.base = 'c' and h.hz = 3 and h.ret = 1 then 1 else 0 end as r3n, case when h.base = 'c' and h.hz = 3 then 1 else 0 end as r3d, case when h.base = 'c' and h.hz = 6 and h.ret = 1 then 1 else 0 end as r6n, case when h.base = 'c' and h.hz = 6 then 1 else 0 end as r6d, case when h.base = 'a' and h.hz = 3 and h.ret = 1 then 1 else 0 end as r3an, case when h.base = 'a' and h.hz = 3 then 1 else 0 end as r3ad, case when h.base = 'a' and h.hz = 6 and h.ret = 1 then 1 else 0 end as r6an, case when h.base = 'a' and h.hz = 6 then 1 else 0 end as r6ad, 0 as hch, 0 as hirea
from (
    select q.*,
           case when q.fire_dt is null or q.fire_dt >= q.md then 1 else 0 end as ret
    from (
        select h.*, z.grain as zgrain, z.hz,
               (case when z.mons is not null then h.hire_dt + z.mons * interval '1 month'
                     else h.hire_dt + z.dys * interval '1 day' end)::date as md
        from hh_cohort h
        cross join (values ('m'::text, 3, 3, null::int), ('m'::text, 6, 6, null::int),
                           ('w'::text, 3, null::int, 91), ('w'::text, 6, null::int, 182)) z(grain, hz, mons, dys)
    ) q
) h
join hh_calendar c on c.grain = h.zgrain
    and (case when h.zgrain = 'm' then h.hire_dt else h.md end) between c.slot_start and c.slot_end
cross join hh_param p
where h.md <= p.data_dt
distributed by (path_s);

analyze hh_evt;
```

## 10 HH · куб по разрезам — заменить код целиком (поставка 44: закрепляемость — и дозревающие месяцы найма, комментарий; поставка 43: когорта месяца найма без окна; поставка 42: найм с начала календарного года; поставка 39: nrd в кубе и базе — колонки m_nrd, w_nrd сразу после m_nr, w_nr)

```sql
-- ============================================================================
-- HH · куб по разрезам → hrbp_hub_cube. Ключ строки — путь юнита × 6 разрезов
-- численности; значения — массивы по слотам: m_* 24 месяца, w_* 24 недели
-- (позиция = idx из hrbp_hub_calendar). Проценты не храним — только числитель и
-- знаменатель, поэтому любую строку можно складывать с любой другой.
--
-- ОКНА (меняются здесь и только здесь):
--   rg, nrg (regrettable / non-regrettable увольнения) — 12 мес / 52 нед;
--   hcw — знаменатель текучести, всегда по АКТИВНОЙ численности HQ, какие бы разрезы ни были выбраны
--   (HRBP 13.10, Дьяченко): месяцы — численность активной HQ на конец месяца (hch, без окна: из неё чарт
--   считает текучесть с начала года), недели — Σ за 52 нед (как раньше; в отчёте не показываются);
--   nr (незаполненные причины) — с начала года (месяцы: окно wy — с января того же года; недели — 13 нед):
--   увольнения с 1 января, у которых прошло 30+ дней, а причины нет (HRBP 13.10);
--   nrd — знаменатель nr тем же окном: все увольнения с 1 января, у которых прошло 30+ дней (Дьяченко 16.10);
--   hirea — найм в активную численность (без окна): найм в компанию с типом «Активная» + переходы
--   кандидатов + внутренние переводы (candidate_transfer_flg, internal_transfer_flg; HRBP 13.10);
--   r3*/r6* (когорта найма) — месяцы (Дьяченко 19.10, было — когорта с начала года по слоту созревания): БЕЗ окна,
--          в слоте МЕСЯЦА НАЙМА. Знаменатель — все нанятые в этом месяце, числитель — те из них, кто работает
--          через 3 (6) мес от найма — только те, у кого срок уже прошёл (найм + 3 (6) ≤ data_dt). Месяц, где срок
--          прошёл у всех (конец месяца «найм + 3 (6)» ≤ data_dt), — созревший; остальные — дозревающие (в чарте —
--          пунктиром). Средняя с начала года (Σ числителей / Σ знаменателей созревших месяцев с января) — в чарте. Недели — как было: 39 и 26 нед по слоту созревания (в
--          отчёте не показываются).
--   hc, jun, hire, fire — без окна (запас на конец слота и поток слота).
--   rgs, nrgs — те же regrettable / non-regrettable увольнения БЕЗ окна (поток слота): по ним чарт
--               считает текучесть нарастающим итогом с начала года (владелец 12.10).
-- Окно считается на каждой строке куба: сумма окон по строкам = окно суммы.
-- ============================================================================
drop table if exists hh_cube_long;
create table hh_cube_long as
select md5(path_s || '|' || paint || '|' || it || '|' || stream || '|' || spec || '|' || staff || '|' || hct) as kid,
       path_s, paint, it, stream, spec, staff, hct, grain, idx,
       sum(hc) as hc,
       sum(jun) as jun,
       sum(fire) as fire,
       sum(rg) as rg,
       sum(nrg) as nrg,
       sum(nr) as nr,
       sum(nrd) as nrd,
       sum(hire) as hire,
       sum(r3n) as r3n,
       sum(r3d) as r3d,
       sum(r6n) as r6n,
       sum(r6d) as r6d,
       sum(r3an) as r3an,
       sum(r3ad) as r3ad,
       sum(r6an) as r6an,
       sum(r6ad) as r6ad,
       sum(hch) as hch,
       sum(hirea) as hirea
from hh_evt
group by path_s, paint, it, stream, spec, staff, hct, grain, idx
distributed by (kid);

-- Плотная сетка «ключ × все слоты», иначе окно в строках захватит не те слоты.
drop table if exists hh_cube_win;
create table hh_cube_win as
select x.kid, x.grain, x.idx,
       x.hc,
       x.jun,
       case when x.grain = 'm' then sum(x.rg) over w12 else sum(x.rg) over w52 end as rg,
       case when x.grain = 'm' then sum(x.nrg) over w12 else sum(x.nrg) over w52 end as nrg,
       x.rg as rgs,
       x.nrg as nrgs,
       case when x.grain = 'm' then x.hch else sum(x.hch) over w52 end as hcw,
       case when x.grain = 'm' then sum(x.nr) over wy else sum(x.nr) over w13 end as nr,
       case when x.grain = 'm' then sum(x.nrd) over wy else sum(x.nrd) over w13 end as nrd,
       case when x.grain = 'm' then x.r3n else sum(x.r3n) over w39 end as r3n,
       case when x.grain = 'm' then x.r3d else sum(x.r3d) over w39 end as r3d,
       case when x.grain = 'm' then x.r6n else sum(x.r6n) over w26 end as r6n,
       case when x.grain = 'm' then x.r6d else sum(x.r6d) over w26 end as r6d,
       case when x.grain = 'm' then x.r3an else sum(x.r3an) over w39 end as r3an,

       case when x.grain = 'm' then x.r6ad else sum(x.r6ad) over w26 end as r6ad,
       x.hirea,
       x.hire,
       x.fire
from (
    select g.kid, g.grain, g.idx,
           coalesce(l.hc, 0) as hc,

                   coalesce(l.r3an, 0) as r3an,
                   coalesce(l.r3ad, 0) as r3ad,
                   coalesce(l.r6an, 0) as r6an,
                   coalesce(l.r6ad, 0) as r6ad,
                   coalesce(l.hch, 0) as hch,
                   coalesce(l.hirea, 0) as hirea
    from (select k.kid, c.grain, c.idx
          from (select distinct kid from hh_cube_long) k
          cross join hh_calendar c) g
    left join hh_cube_long l
        on l.kid = g.kid and l.grain = g.grain and l.idx = g.idx
) x
window w12 as (partition by x.kid, x.grain order by x.idx rows between 11 preceding and current row),
       wy as (partition by x.kid, x.grain, floor(x.idx / 12.0) order by x.idx rows between unbounded preceding and current row),
       w52 as (partition by x.kid, x.grain order by x.idx rows between 51 preceding and current row),
       w3 as (partition by x.kid, x.grain order by x.idx rows between 2 preceding and current row),
       w13 as (partition by x.kid, x.grain order by x.idx rows between 12 preceding and current row),
       w39 as (partition by x.kid, x.grain order by x.idx rows between 38 preceding and current row),
       w26 as (partition by x.kid, x.grain order by x.idx rows between 25 preceding and current row)
distributed by (kid);
```

```sql
from (
    select e.*, a.attr_k,
           coalesce(case a.attr_k when 'grade' then e.a_grade when 'seniority' then e.seniority when 'exp' then e.a_exp when 'gender' then e.a_gender when 'age' then e.a_age when 'office' then e.a_office when 'work' then e.a_work when 'head' then e.a_head when 'legal' then e.a_legal when 'macro' then e.a_macro when 'city' then e.a_city end, '-') as attr_v
    from hh_evt e
    cross join (values ('grade'), ('seniority'), ('exp'), ('gender'), ('age'), ('office'), ('work'), ('head'), ('legal'), ('macro'), ('city')) a(attr_k)
    where e.grain = 'm'
) t
group by path_s, paint, it, stream, spec, staff, hct, attr_k, attr_v, idx
distributed by (kid);

-- Свёртка — строками того же формата: ключ 'T:<id юнита>', разрезы пустые. Дальше она идёт
-- теми же окнами и массивами, что и листовые строки (окна суммируются, поэтому окно суммы
-- = сумма окон), и в конце уходит в свою таблицу.
insert into hh_attr_long
select md5('T:' || u.id || '|' || l.attr_k || '|' || l.attr_v) as kid,
       'T:' || u.id as path_s, '' as paint, '' as it, '' as stream, '' as spec, '' as staff, '' as hct,
       l.attr_k, l.attr_v, l.idx,
       sum(l.hc) as hc,
       sum(l.jun) as jun,
       sum(l.fire) as fire,
       sum(l.rg) as rg,
       sum(l.nrg) as nrg,
       sum(l.nr) as nr,
       sum(l.nrd) as nrd,
       sum(l.hire) as hire,
       sum(l.r3n) as r3n,
       sum(l.r3d) as r3d,
       sum(l.r6n) as r6n,
       sum(l.r6d) as r6d,
       sum(l.r3an) as r3an,
       sum(l.r3ad) as r3ad,
       sum(l.r6an) as r6an,
       sum(l.r6ad) as r6ad,
       sum(l.hch) as hch,
       sum(l.hirea) as hirea
from hh_attr_long l
join (select q.path_s, unnest(string_to_array(q.path_s, '/')) as uid
      from (select distinct path_s from hh_attr_long) q) p
    on p.path_s = l.path_s
join hrbp_hub_unit u
    on u.id = p.uid and u.attr_top = 1
cross join hh_param pr
where l.hct <> pr.nonlegal_type  -- свёртка — «без разрезов», а без разрезов отчёт берёт только юридическую численность
group by u.id, l.attr_k, l.attr_v, l.idx;

drop table if exists hh_attr_win;
create table hh_attr_win as
select x.kid, x.grain, x.idx,
       x.hc,
       x.jun,
       sum(x.rg) over w12 as rg,
       sum(x.nrg) over w12 as nrg,
       x.rg as rgs,
       x.nrg as nrgs,
       x.hch as hcw,
       sum(x.nr) over wy as nr,
       sum(x.nrd) over wy as nrd,
       x.r3n as r3n,
       x.r3d as r3d,
       x.r6n as r6n,
       x.r6d as r6d,
       x.r3an as r3an,
       x.r3ad as r3ad,
       x.r6an as r6an,
       x.r6ad as r6ad,
       x.hirea,
       x.hire,
       x.fire
from (
    select g.kid, g.grain, g.idx,
           coalesce(l.hc, 0) as hc,
                   coalesce(l.jun, 0) as jun,
                   coalesce(l.fire, 0) as fire,
                   coalesce(l.rg, 0) as rg,
                   coalesce(l.nrg, 0) as nrg,
                   coalesce(l.nr, 0) as nr,
                   coalesce(l.nrd, 0) as nrd,
                   coalesce(l.hire, 0) as hire,
                   coalesce(l.r3n, 0) as r3n,
                   coalesce(l.r3d, 0) as r3d,
                   coalesce(l.r6n, 0) as r6n,
                   coalesce(l.r6d, 0) as r6d,
                   coalesce(l.r3an, 0) as r3an,
                   coalesce(l.r3ad, 0) as r3ad,
                   coalesce(l.r6an, 0) as r6an,
                   coalesce(l.r6ad, 0) as r6ad,
                   coalesce(l.hch, 0) as hch,
                   coalesce(l.hirea, 0) as hirea
    from (select k.kid, c.grain, c.idx
          from (select distinct kid from hh_attr_long) k
          cross join hh_calendar c
          where c.grain = 'm') g
    left join hh_attr_long l
        on l.kid = g.kid and l.idx = g.idx
) x
window w12 as (partition by x.kid, x.grain order by x.idx rows between 11 preceding and current row),
       wy as (partition by x.kid, x.grain, floor(x.idx / 12.0) order by x.idx rows between unbounded preceding and current row),
       w3 as (partition by x.kid, x.grain order by x.idx rows between 2 preceding and current row)
distributed by (kid);

drop table if exists hh_attr_arr;
create table hh_attr_arr as
select kid,
       array_agg(hc::int order by idx) as hc,
       array_agg(jun::int order by idx) as jun,
       array_agg(rg::int order by idx) as rg,
       array_agg(nrg::int order by idx) as nrg,
       array_agg(rgs::int order by idx) as rgs,
       array_agg(nrgs::int order by idx) as nrgs,
       array_agg(hcw::int order by idx) as hcw,
       array_agg(nr::int order by idx) as nr,
       array_agg(nrd::int order by idx) as nrd,
       array_agg(r3n::int order by idx) as r3n,
       array_agg(r3d::int order by idx) as r3d,
       array_agg(r6n::int order by idx) as r6n,
       array_agg(r6d::int order by idx) as r6d,
       array_agg(r3an::int order by idx) as r3an,
       array_agg(r3ad::int order by idx) as r3ad,
       array_agg(r6an::int order by idx) as r6an,
       array_agg(r6ad::int order by idx) as r6ad,
       array_agg(hire::int order by idx) as hire,
       array_agg(fire::int order by idx) as fire,
       array_agg(hirea::int order by idx) as hirea,
       sum(abs(hc)) + sum(abs(jun)) + sum(abs(rg)) + sum(abs(nrg)) + sum(abs(hcw)) + sum(abs(nr)) + sum(abs(nrd)) + sum(abs(r3n)) + sum(abs(r3d)) + sum(abs(r6n)) + sum(abs(r6d)) + sum(abs(r3an)) + sum(abs(r3ad)) + sum(abs(r6an)) + sum(abs(r6ad)) + sum(abs(hire)) + sum(abs(fire)) + sum(abs(hirea)) as tot
from hh_attr_win
where grain = 'm' and idx >= 0
group by kid
distributed by (kid);
```
